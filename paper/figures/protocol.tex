% Fig. -- flowchart of the nested leave-one-city-out protocol.
\begin{tikzpicture}[
  font=\footnotesize, node distance=4mm,
  blk/.style={draw, rectangle, minimum height=7mm, text width=58mm, align=center,
               inner sep=2pt, line width=0.5pt},
  term/.style={blk, rounded corners=3mm, text width=30mm, fill=tnblue!12},
  dec/.style={draw, diamond, aspect=3, align=center, inner sep=1pt, line width=0.5pt,
              text width=24mm},
  arr/.style={-{Stealth[length=1.8mm,width=1.5mm]}, line width=0.5pt}]
\node[term] (start) {Corpora of $n$ cities};
\node[blk, below=of start] (fold) {Hold out target city $c$; the other $n-1$ cities are the training set};
\node[blk, below=of fold] (inner) {Inner loop: hold out each training city in turn, fit on the rest, score the standardised RMSE of Eq.~\eqref{eq:selloss}};
\node[blk, below=of inner] (pick) {Keep the grid value with the smallest mean inner error; number of rounds or epochs = median best iteration};
\node[blk, below=of pick] (refit) {Refit on all $n-1$ training cities (five seeds); predict tile labels and links of city $c$};
\node[blk, below=of refit] (eval) {Score on the ray-traced truth of $c$: path loss, serving site, SINR, coverage, cell edges, planning};
\node[dec, below=of eval] (more) {All $n$ cities\\held out?};
\node[blk, below=of more] (stats) {City-level inference: bootstrap intervals, exact Wilcoxon tests, Holm correction; certificate from the $n$ residual sets};
\node[term, below=of stats] (end) {Reported results};
\draw[arr] (start) -- (fold);
\draw[arr] (fold) -- (inner);
\draw[arr] (inner) -- (pick);
\draw[arr] (pick) -- (refit);
\draw[arr] (refit) -- (eval);
\draw[arr] (eval) -- (more);
\draw[arr] (more) -- node[right, font=\scriptsize] {yes} (stats);
\draw[arr] (more.west) -- ++(-5mm,0) node[above, font=\scriptsize, xshift=-1mm] {no} |- (fold.west);
\draw[arr] (stats) -- (end);
\end{tikzpicture}
